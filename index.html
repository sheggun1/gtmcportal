async function parentAsk() { const question = q('pQuestion').value.trim(); if (!question) return toast('Type a question', 'err'); const idx = q('pWard').value; if (idx === '') return toast('Select a ward first', 'err'); const ward = (window.__WARDS||[])[Number(idx)]; if (!ward) return toast('Ward selection is unavailable', 'err'); const chat = q('pChat'); chat.innerHTML = `
You:\\n${esc(question)}
Thinking…
`; try { const resp = await api('parentAskResultAI', { className: ward.className, studentId: ward.studentId, generalMode: true, question }); chat.innerHTML = `
You:\\n${esc(question)}
Assistant:\\n${esc(resp.answer)}
`; q('pQuestion').value = ''; } catch (e) { chat.innerHTML = `
${esc(e.message)}
`; } }
